rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /examens/debogage_hiver_2026/codes/{codeId} {

      allow list: if false;

      allow get: if codeId.matches('^[0-9]{10}$');

      allow create: if false;
      allow delete: if false;

      allow update: if
        resource.data.status == "available"
        && request.resource.data.status == "used"
        && request.resource.data.code == resource.data.code
        && request.resource.data.examId == resource.data.examId
        && request.resource.data.usedAt == request.time
        && request.resource.data.usedBySessionId is string
        && request.resource.data.verificationFingerprint is string
        && request.resource.data.verificationSource == "debogage-h2026-firebase-proof-v1"
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "status",
          "usedAt",
          "usedBySessionId",
          "verificationFingerprint",
          "verificationSource"
        ]);
    }

    match /examens/debogage_hiver_2026/accessLogs/{sessionId} {

      allow read: if false;
      allow list: if false;
      allow update: if false;
      allow delete: if false;

      allow create: if
        sessionId == request.resource.data.sessionId
        && request.resource.data.keys().hasOnly([
          "code",
          "examId",
          "studentName",
          "studentId",
          "studentGroup",
          "sessionId",
          "createdAt",
          "activationClientIso",
          "activationClientText",
          "verificationFingerprint",
          "verificationSource",
          "userAgent"
        ])
        && request.resource.data.examId == "debogage_hiver_2026"
        && request.resource.data.code.matches('^[0-9]{10}$')
        && request.resource.data.studentName is string
        && request.resource.data.studentId is string
        && request.resource.data.studentGroup is string
        && request.resource.data.sessionId is string
        && request.resource.data.activationClientIso is string
        && request.resource.data.activationClientText is string
        && request.resource.data.verificationFingerprint is string
        && request.resource.data.verificationSource == "debogage-h2026-firebase-proof-v1"
        && request.resource.data.createdAt == request.time;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
